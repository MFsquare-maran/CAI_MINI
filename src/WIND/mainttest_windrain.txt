#include <Arduino.h>
#include "config_WIND.h"

constexpr uint32_t SERIAL_DEBUG_BAUD = 115200U;

// Debounce-Zeiten (in Mikrosekunden)
constexpr uint32_t RAIN_DEBOUNCE_US = 300000;  // 10 ms für Regenwippe
constexpr uint32_t WIND_DEBOUNCE_US = 1000;   // 1 ms für Anemometer

// Counter
volatile uint32_t rain_count = 0;
volatile uint32_t wind_count = 0;

// Debounce-Zeitstempel
volatile uint32_t last_rain_us = 0;
volatile uint32_t last_wind_us = 0;

// Zähler wie viele Pulse verworfen wurden (zur Kontrolle)
volatile uint32_t rain_bounced = 0;
volatile uint32_t wind_bounced = 0;

// ── ISRs ────────────────────────────────────────────────
void IRAM_ATTR isr_rain() {
    uint32_t now = micros();
    if (now - last_rain_us < RAIN_DEBOUNCE_US) {
        rain_bounced++;
        return;
    }
    last_rain_us = now;
    rain_count++;
}

void IRAM_ATTR isr_wind() {
    uint32_t now = micros();
    if (now - last_wind_us < WIND_DEBOUNCE_US) {
        wind_bounced++;
        return;
    }
    last_wind_us = now;
    wind_count++;
}

// ── Setup ───────────────────────────────────────────────
void setup() {
    Serial.begin(SERIAL_DEBUG_BAUD);
    delay(3000);

    Serial.println("╔══════════════════════════════╗");
    Serial.println("║   Interrupt-Test mit 10 MHz  ║");
    Serial.println("╚══════════════════════════════╝");

    // CPU auf 10 MHz runtertakten
    

    Serial.print("CPU Frequency: ");
    Serial.print(getCpuFrequencyMhz());
    Serial.println(" MHz");

    Serial.print("Rain Debounce: ");
    Serial.print(RAIN_DEBOUNCE_US / 1000);
    Serial.println(" ms");
    Serial.print("Wind Debounce: ");
    Serial.print(WIND_DEBOUNCE_US / 1000);
    Serial.println(" ms");

    pinMode(WIND_SPEED, INPUT_PULLDOWN);
    pinMode(RAIN_GAUGE, INPUT_PULLDOWN);

    attachInterrupt(digitalPinToInterrupt(WIND_SPEED), isr_wind, RISING);
    attachInterrupt(digitalPinToInterrupt(RAIN_GAUGE), isr_rain, RISING);

    Serial.println("Bereit. Wippe / Anemometer auslösen...");

    setCpuFrequencyMhz(10);
    
}

// ── Loop ────────────────────────────────────────────────
void loop() {
    Serial.print("Wind: ");
    Serial.print(wind_count);
    Serial.print(" (bounced: ");
    Serial.print(wind_bounced);
    Serial.print(")   Rain: ");
    Serial.print(rain_count);
    Serial.print(" (bounced: ");
    Serial.print(rain_bounced);
    Serial.println(")");
    delay(1000);
}
